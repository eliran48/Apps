איך להגדיר את שם הקמפיין?

בשורה 8 בקוד למטה תוכל לרשום את שם הקמפיין הרצוי, למשל: var MANUAL_CAMPAIGN_NAME = "קמפיין תמיכה 2026";.
אם תשאיר אותה ריקה (""), הקוד ייקח אוטומטית את שם קובץ ה-Google Sheet! 

function sendLeadToFirebase(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();
    
    // 👈 רשום כאן את שם הקמפיין הרצוי. אם תשאיר ריק (""), הקוד ייקח את שם הקובץ אוטומטית!
    var MANUAL_CAMPAIGN_NAME = ""; 

    var autoCampaign = ss.getName().replace(/\s*\(תגובות\)\s*/i, "").replace(/\s*\(Responses\)\s*/i, "").trim();
    var campaignName = MANUAL_CAMPAIGN_NAME.trim() || autoCampaign || sheet.getName() || "קמפיין גוגל שיטס";
    var sheetName = sheet.getName() || "גיליון 1";

    var row = (e && e.range) ? e.range.getRow() : sheet.getLastRow();
    if (row <= 1) return;

    var lastCol = sheet.getLastColumn();
    if (lastCol < 1) return;

    var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    var values = sheet.getRange(row, 1, 1, lastCol).getValues()[0];

    var name = "", phone = "", email = "";
    var maritalStatus = "", gamblingStatus = "", lgbtq = "";
    var extraDetails = {};

    for (var i = 0; i < headers.length; i++) {
      var headerRaw = String(headers[i] || "").trim();
      var headerClean = headerRaw.toLowerCase().replace(/_/g, " ");
      var val = (values[i] !== undefined && values[i] !== null) ? String(values[i]).trim() : "";

      if (!val) continue;

      if (headerClean === "first name" || headerClean.indexOf("first_name") !== -1 || headerClean.indexOf("שם") !== -1 || headerClean.indexOf("name") !== -1) {
        if (!name) name = val;
      } else if (headerClean === "phone number" || headerClean.indexOf("phone_number") !== -1 || headerClean.indexOf("טלפון") !== -1 || headerClean.indexOf("phone") !== -1 || headerClean.indexOf("נייד") !== -1) {
        if (!phone) phone = val;
      } else if (headerClean.indexOf("מייל") !== -1 || headerClean.indexOf("email") !== -1) {
        if (!email) email = val;
      } else if (headerClean.indexOf("זוגי") !== -1 || headerClean.indexOf("סטטוס") !== -1) {
        if (!maritalStatus) maritalStatus = val;
      } else if (headerClean.indexOf("הימורים") !== -1 || headerClean.indexOf("מצב") !== -1) {
        if (!gamblingStatus) gamblingStatus = val;
      } else if (headerClean.indexOf("להט") !== -1 || headerClean.indexOf("lgbt") !== -1) {
        if (!lgbtq) lgbtq = val;
      } else {
        extraDetails[headerRaw] = val;
      }
    }

    if (!name && values.length > 0) name = String(values[0] || values[1] || "");
    if (!phone && values.length > 1) phone = String(values[1] || values[2] || "");

    // בניית ה-JSON בפורמט מאובטח שמבטיח שכל השדות מחזירים מחרוזת תקינה
    var payload = {
      fields: {
        name: { stringValue: String(name || "פנייה משיטס") },
        phone: { stringValue: String(phone || "") },
        email: { stringValue: String(email || "") },
        source: { stringValue: String(campaignName || "קמפיין שיטס") },
        sheetName: { stringValue: String(sheetName || "גיליון 1") },
        maritalStatus: { stringValue: String(maritalStatus || "") },
        gamblingStatus: { stringValue: String(gamblingStatus || "") },
        lgbtq: { stringValue: String(lgbtq || "") },
        extraDetailsJson: { stringValue: String(JSON.stringify(extraDetails) || "{}") },
        status: { stringValue: "חדש" },
        timestamp: { stringValue: new Date().toISOString() }
      }
    };
    
    var url = "https://firestore.googleapis.com/v1/projects/eliran-apps/databases/(default)/documents/artifacts/eliran-apps/public/data/leads";
    
    var options = {
      method: "post",
      contentType: "application/json",
      payload: JSON.stringify(payload),
      muteHttpExceptions: true
    };
    
    var response = UrlFetchApp.fetch(url, options);
    Logger.log("Response Code: " + response.getResponseCode());
    Logger.log("Response Body: " + response.getContentText());
  } catch (err) {
    Logger.log("Error in sendLeadToFirebase: " + err.toString());
  }
}
