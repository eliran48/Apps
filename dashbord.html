var MANUAL_CAMPAIGN_NAME = ""; 

function sendLeadToFirebase(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = (e && e.range) ? e.range.getSheet() : ss.getActiveSheet();
    var row = (e && e.range) ? e.range.getRow() : sheet.getLastRow();
    if (row <= 1) return;
    sendRowToFirebase(ss, sheet, row);
  } catch (err) {
    Logger.log("Error: " + err.toString());
  }
}

function syncAllExistingLeads() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var totalCount = 0;
    for (var s = 0; s < sheets.length; s++) {
      var sheet = sheets[s];
      if (sheet.isSheetHidden()) continue;
      var lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        for (var r = 2; r <= lastRow; r++) {
          if (sendRowToFirebase(ss, sheet, r)) totalCount++;
        }
      }
    }
    SpreadsheetApp.getUi().alert("הסנכרון הושלם! סונכרנו " + totalCount + " לידים לדאשבורד.");
  } catch (err) {
    SpreadsheetApp.getUi().alert("שגיאה בסנכרון: " + err.toString());
  }
}

function sendRowToFirebase(ss, sheet, row) {
  var autoCampaign = ss.getName().replace(/\s*\(תגובות\)\s*/i, "").replace(/\s*\(Responses\)\s*/i, "").trim();
  var campaignName = MANUAL_CAMPAIGN_NAME.trim() || autoCampaign || sheet.getName() || "קמפיין שיטס";
  var sheetName = sheet.getName() || "גיליון 1";
  var lastCol = sheet.getLastColumn();
  if (lastCol < 1) return false;

  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var rawValues = sheet.getRange(row, 1, 1, lastCol).getValues()[0];
  var displayValues = sheet.getRange(row, 1, 1, lastCol).getDisplayValues()[0];

  var firstName = "", lastName = "", fullName = "", phone = "", email = "", notes = "", sheetStatus = "";
  var maritalStatus = "", gamblingStatus = "", lgbtq = "", extraDetails = {}, rowDate = null;

  for (var i = 0; i < headers.length; i++) {
    var headerRaw = String(headers[i] || "").trim();
    var headerClean = headerRaw.toLowerCase().replace(/_/g, " ").replace(/-/g, " ");
    var val = (displayValues[i] !== undefined && String(displayValues[i]).trim() !== "") ? String(displayValues[i]).trim() : ((rawValues[i] !== undefined) ? String(rawValues[i]).trim() : "");

    if (!val) continue;

    if (headerClean.indexOf("חותמת") !== -1 || headerClean.indexOf("תאריך") !== -1 || headerClean.indexOf("זמן") !== -1 || headerClean.indexOf("timestamp") !== -1 || headerClean.indexOf("date") !== -1) {
      if (!rowDate) rowDate = parseSheetDate(rawValues[i], displayValues[i]);
    } else if (headerClean.indexOf("שם פרטי") !== -1 || headerClean.indexOf("first_name") !== -1) {
      firstName = val;
    } else if (headerClean.indexOf("שם משפחה") !== -1 || headerClean.indexOf("last_name") !== -1) {
      lastName = val;
    } else if (!fullName && (headerClean.indexOf("שם") !== -1 || headerClean.indexOf("name") !== -1)) {
      fullName = val;
    } else if (headerClean.indexOf("טלפון") !== -1 || headerClean.indexOf("נייד") !== -1 || headerClean.indexOf("סלולרי") !== -1 || headerClean.indexOf("phone") !== -1) {
      if (!phone) phone = val;
    } else if (headerClean.indexOf("מייל") !== -1 || headerClean.indexOf("אימייל") !== -1 || headerClean.indexOf("דוא\"ל") !== -1 || headerClean.indexOf("email") !== -1) {
      if (!email) email = val;
    } else if (headerClean.indexOf("הערות") !== -1 || headerClean.indexOf("הערה") !== -1 || headerClean.indexOf("note") !== -1) {
      notes = val;
    } else if (headerClean.indexOf("סטטוס") !== -1 || headerClean.indexOf("status") !== -1) {
      sheetStatus = val;
    } else {
      extraDetails[headerRaw] = val;
    }
  }

  var name = (firstName || lastName) ? (firstName + " " + lastName).trim() : fullName;
  if (!rowDate) rowDate = new Date();
  var safeIsoDate = (rowDate && typeof rowDate.toISOString === 'function') ? rowDate.toISOString() : new Date().toISOString();

  if (!phone && !name && !email) return false;

  var docId = (ss.getId() + "_" + sheet.getSheetId() + "_row_" + row).replace(/[^a-zA-Z0-9_-]/g, "_");

  var payload = {
    fields: {
      name: { stringValue: String(name || "פנייה משיטס") },
      phone: { stringValue: String(phone || "") },
      email: { stringValue: String(email || "") },
      source: { stringValue: String(campaignName) },
      sheetName: { stringValue: String(sheetName) },
      maritalStatus: { stringValue: String(maritalStatus) },
      gamblingStatus: { stringValue: String(gamblingStatus) },
      lgbtq: { stringValue: String(lgbtq) },
      notes: { stringValue: String(notes) },
      extraDetailsJson: { stringValue: JSON.stringify(extraDetails) },
      status: { stringValue: String(sheetStatus || "חדש") },
      timestamp: { stringValue: safeIsoDate }
    }
  };

  var apiKey = "AIzaSyCotYARE-PfM2vPdr3UtRzVu5M_1FieYKk";
  var url = "https://firestore.googleapis.com/v1/projects/eliran-apps/databases/(default)/documents/artifacts/eliran-apps/public/data/leads/" + docId + "?key=" + apiKey;

  var options = {
    method: "patch",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  var res = UrlFetchApp.fetch(url, options);
  return (res.getResponseCode() >= 200 && res.getResponseCode() < 300);
}

function parseSheetDate(rawVal, displayVal) {
  if (rawVal && (Object.prototype.toString.call(rawVal) === '[object Date]' || typeof rawVal.getTime === 'function')) {
    return new Date(rawVal.getTime());
  }
  var str = String(displayVal || rawVal || "").trim();
  var d = new Date(str);
  return !isNaN(d.getTime()) ? d : null;
}
