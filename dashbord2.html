הוראות פשוטות לחיבור קבצים וטפסים חדשים:

  1. **שימוש באותו קוד בדיוק**: הדבק את הקוד למטה בעורך ה-Apps Script בכל קובץ Google Sheets שברצונך לחבר.
  2. **משיכה אוטומטית של שדות מיוחדים**: הקוד מזהה עמודות רגילות (שם, טלפון, מייל, הערות, סטטוס), ו**כל שאר העמודות והשאלות המיוחדות** מנזילות אוטומטית ל-"שדות נוספים" המוצגים בדאשבורד בלחיצה על העין (👁️)!
  3. **הגדרת טריגר לעדכון אוטומטי בלייב**:
    • בסרגל הצדדי ב-Apps Script לחץ על **סמל השעון ⏰ (טריגרים / Triggers)**.
    • לחץ למטה על **"הוספת טריגר" (Add Trigger)**.
    • בחירת פונקציה: sendLeadToFirebase.
    • בחירת סוג האירוע: שנה ל-**בעת שליחת טופס (On form submit)** ולחץ שמירה.

// 👈 רשום כאן את שם הקמפיין הרצוי. אם תשאיר ריק (""), הקוד ייקח את שם קובץ ה-Google Sheet אוטומטית!
var MANUAL_CAMPAIGN_NAME = ""; 

// 1. פונקציה לשליחה אוטומטית בעת קבלת ליד חדש (On Form Submit)
function sendLeadToFirebase(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = (e && e.range) ? e.range.getSheet() : ss.getActiveSheet();
    var row = (e && e.range) ? e.range.getRow() : sheet.getLastRow();
    if (row <= 1) return;
    
    sendRowToFirebase(ss, sheet, row);
  } catch (err) {
    Logger.log("Error in sendLeadToFirebase: " + err.toString());
  }
}

// 2. פונקציה לסנכרון חכם של כל הלידים מכל הגיליונות בקובץ (ללא כפילויות!)
function syncAllExistingLeads() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var totalCount = 0;
    
    for (var s = 0; s < sheets.length; s++) {
      var sheet = sheets[s];
      if (sheet.isSheetHidden()) continue; // דילוג על גיליונות מוסתרים
      
      var lastRow = sheet.getLastRow();
      if (lastRow > 1) {
        for (var r = 2; r <= lastRow; r++) {
          var sent = sendRowToFirebase(ss, sheet, r);
          if (sent) totalCount++;
        }
      }
    }
    
    if (totalCount === 0) {
      SpreadsheetApp.getUi().alert("לא נמצאו שורות לידים תקינות לסנכרון בגיליונות אלו.");
    } else {
      SpreadsheetApp.getUi().alert("הסנכרון הושלם בהצלחה! מסונכרנים " + totalCount + " לידים תקינים מכלל הגיליונות בקובץ זה לדאשבורד.");
    }
  } catch (err) {
    SpreadsheetApp.getUi().alert("שגיאה בסנכרון: " + err.toString());
  }
}

function clearFirebaseDatabase() {
  try {
    var apiKey = "AIzaSyCotYARE-PfM2vPdr3UtRzVu5M_1FieYKk";
    var url = "https://firestore.googleapis.com/v1/projects/eliran-apps/databases/(default)/documents/artifacts/eliran-apps/public/data/leads?key=" + apiKey;
    var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (res.getResponseCode() !== 200) {
      SpreadsheetApp.getUi().alert("מסד הנתונים ב-Firebase כבר ריק (0 לידים).");
      return;
    }
    var data = JSON.parse(res.getContentText());
    if (!data.documents || data.documents.length === 0) {
      SpreadsheetApp.getUi().alert("מסד הנתונים ב-Firebase כבר ריק (0 לידים).");
      return;
    }
    var count = 0;
    for (var i = 0; i < data.documents.length; i++) {
      var docPath = data.documents[i].name;
      var delUrl = "https://firestore.googleapis.com/v1/" + docPath + "?key=" + apiKey;
      UrlFetchApp.fetch(delUrl, { method: "delete", muteHttpExceptions: true });
      count++;
    }
    SpreadsheetApp.getUi().alert("בוצע ניקוי מלא ב-Firebase! נמחקו " + count + " לידים בענן.");
  } catch (err) {
    SpreadsheetApp.getUi().alert("שגיאה בניקוי: " + err.toString());
  }
}

// פונקציה לפענוח תאריכים מתוך הגוגל שיטס
function parseSheetDate(val) {
  if (!val) return null;
  if (val instanceof Date) return isNaN(val.getTime()) ? null : val;
  
  var d = new Date(val);
  if (!isNaN(d.getTime())) return d;

  var str = String(val).trim();
  var parts = str.split(/[\s,T]+/);
  if (parts.length >= 1) {
    var dateParts = parts[0].split(/[\/\.-]/);
    if (dateParts.length === 3) {
      var day = parseInt(dateParts[0], 10);
      var month = parseInt(dateParts[1], 10) - 1;
      var year = parseInt(dateParts[2], 10);
      if (year < 100) year += 2000;
      var hours = 0, minutes = 0, seconds = 0;
      if (parts.length >= 2) {
        var timeParts = parts[1].split(":");
        if (timeParts.length >= 2) {
          hours = parseInt(timeParts[0], 10);
          minutes = parseInt(timeParts[1], 10);
          if (timeParts.length >= 3) seconds = parseInt(timeParts[2], 10);
        }
      }
      var parsed = new Date(year, month, day, hours, minutes, seconds);
      if (!isNaN(parsed.getTime())) return parsed;
    }
  }
  return null;
}

// פונקציית עזר לשליחה ועדכון שורה בודדת במזהה ייחודי
function sendRowToFirebase(ss, sheet, row) {
  var autoCampaign = ss.getName().replace(/\s*\(תגובות\)\s*/i, "").replace(/\s*\(Responses\)\s*/i, "").trim();
  var campaignName = MANUAL_CAMPAIGN_NAME.trim() || autoCampaign || sheet.getName() || "קמפיין גוגל שיטס";
  var sheetName = sheet.getName() || "גיליון 1";

  var lastCol = sheet.getLastColumn();
  if (lastCol < 1) return false;

  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var values = sheet.getRange(row, 1, 1, lastCol).getValues()[0];

  var name = "", phone = "", email = "", notes = "", sheetStatus = "";
  var maritalStatus = "", gamblingStatus = "", lgbtq = "";
  var extraDetails = {};
  var rowDate = null;

  for (var i = 0; i < headers.length; i++) {
    var headerRaw = String(headers[i] || "").trim();
    var headerClean = headerRaw.toLowerCase().replace(/_/g, " ");
    var val = (values[i] !== undefined && values[i] !== null) ? String(values[i]).trim() : "";

    if (!val) continue;

    // 1. זיהוי תאריך/חותמת זמן
    if (headerClean.indexOf("חותמת") !== -1 || headerClean.indexOf("תאריך") !== -1 || headerClean.indexOf("timestamp") !== -1 || headerClean.indexOf("date") !== -1) {
      if (!rowDate) rowDate = parseSheetDate(values[i]);
    }
    // 2. זיהוי שם (first_name / שם פרטי / שם מלא)
    else if (headerClean === "first name" || headerClean === "first_name" || headerClean === "שם פרטי" || headerClean === "שם מלא") {
      name = val;
    } else if (!name && (headerClean === "שם" || headerClean === "name") && headerClean.indexOf("משפחה") === -1 && headerClean.indexOf("קמפיין") === -1 && headerClean.indexOf("גיליון") === -1) {
      name = val;
    }
    // 3. זיהוי טלפון (phone_number / טלפון / נייד)
    else if (headerClean === "phone number" || headerClean === "phone_number" || headerClean.indexOf("טלפון") !== -1 || headerClean.indexOf("phone") !== -1 || headerClean.indexOf("נייד") !== -1) {
      if (!phone) phone = val;
    }
    // 4. זיהוי מייל
    else if (headerClean.indexOf("מייל") !== -1 || headerClean.indexOf("email") !== -1) {
      if (!email) email = val;
    }
    // 5. זיהוי הערות / הערה / comments
    else if (headerClean.indexOf("הערות") !== -1 || headerClean.indexOf("הערה") !== -1 || headerClean.indexOf("note") !== -1 || headerClean.indexOf("comment") !== -1 || headerClean.indexOf("remark") !== -1) {
      notes = val;
    }
    // 6. זיהוי סטטוס מעודכן בשיטס
    else if (headerClean === "סטטוס" || headerClean === "מצב טיפול" || headerClean === "status") {
      sheetStatus = val;
    }
    // 7. שאלות עומק
    else if (headerClean.indexOf("זוגי") !== -1 || headerClean.indexOf("סטטוס_הזוגי") !== -1) {
      if (!maritalStatus) maritalStatus = val;
    } else if (headerClean.indexOf("הימורים") !== -1 || headerClean.indexOf("מצב_ההימורים") !== -1) {
      if (!gamblingStatus) gamblingStatus = val;
    } else if (headerClean.indexOf("להט") !== -1 || headerClean.indexOf("lgbt") !== -1) {
      if (!lgbtq) lgbtq = val;
    } else {
      // 👈 כל שדה מיוחד אחר בכל טופס מנזיל אוטומטית ל"שדות נוספים"!
      extraDetails[headerRaw] = val;
    }
  }

  if (!rowDate && values.length > 0 && values[0]) {
    rowDate = parseSheetDate(values[0]);
  }
  if (!rowDate) rowDate = new Date();

  // סינון שמות שפוענחו בטעות כתאריכים
  if (name && (name.indexOf(":") !== -1 && name.indexOf("/") !== -1 || /^\d{1,4}[\/\.-]\d{1,2}[\/\.-]\d{1,4}/.test(name))) {
    name = "";
  }

  // סינון שורות ריקות/לא תקינות
  if (!phone && !name && !email) {
    return false;
  }

  var docId = (ss.getId() + "_" + sheet.getSheetId() + "_row_" + row).replace(/[^a-zA-Z0-9_-]/g, "_");

  var payload = {
    fields: {
      name: { stringValue: String(name || "פנייה משיטס") },
      phone: { stringValue: String(phone || "") },
      email: { stringValue: String(email || "") },
      source: { stringValue: String(campaignName || "קמפיין שיטס") },
      sheetName: { stringValue: String(sheetName || "גיליון 1") },
      maritalStatus: { stringValue: String(maritalStatus || "") },
      gamblingStatus: { stringValue: String(gamblingStatus || "") },
      lgbtq: { stringValue: String(lgbtq || "") },
      notes: { stringValue: String(notes || "") },
      extraDetailsJson: { stringValue: String(JSON.stringify(extraDetails) || "{}") },
      status: { stringValue: String(sheetStatus || "חדש") },
      timestamp: { stringValue: rowDate.toISOString() }
    }
  };
  
  var apiKey = "AIzaSyCotYARE-PfM2vPdr3UtRzVu5M_1FieYKk";
  var url = "https://firestore.googleapis.com/v1/projects/eliran-apps/databases/(default)/documents/artifacts/eliran-apps/public/data/leads/" + docId + "?key=" + apiKey;
  
  var options = {
    method: "patch",
    contentType: "application/json",
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };
  
  var response = UrlFetchApp.fetch(url, options);
  Logger.log("DocID: " + docId + " - Response Code: " + response.getResponseCode());
  return true;
}
